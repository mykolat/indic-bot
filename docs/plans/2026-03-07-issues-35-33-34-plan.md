```typescript
export interface DecisionEnvelope {
  maxLeverage: number;
  recommendedLeverage: [number, number]; // [min, max]
  maxSizePct: number;
  recommendedSizePct: [number, number];
  minConfidence: number;
  blockedPairs: string[]; // pairs with existing same-direction positions
  constraints: string[];  // active constraint codes
}

computeEnvelope(
  portfolio: PortfolioState,
  ctx: { fearGreed?: { value: number }; fearGreedLeverageCap?: number; isWeekend?: boolean; weekendLeverageMultiplier?: number; regimeLeverageMultiplier?: number }
): DecisionEnvelope {
  let maxLev = this.config.maxLeverage;
  let maxSize = this.config.maxPositionPct;
  const constraints: string[] = [];

  // F&G cap
  if (ctx.fearGreed && (ctx.fearGreed.value < 25 || ctx.fearGreed.value > 85)) {
    const fgCap = ctx.fearGreedLeverageCap ?? 10;
    maxLev = Math.min(maxLev, fgCap);
    constraints.push('extreme_fear_greed');
  }

  // Session loss scaling
  if (portfolio.sessionPnl < 0 && portfolio.balanceUsd > 0) {
    const lossPct = Math.abs(portfolio.sessionPnl) / portfolio.balanceUsd * 100;
    if (lossPct >= 10) {
      maxLev = Math.min(5, maxLev);
      maxSize = Math.min(25, maxSize);
      constraints.push('session_loss_severe');
    } else if (lossPct >= 5) {
      maxLev = Math.floor(maxLev / 2);
      maxSize = Math.floor(maxSize / 2);
      constraints.push('session_loss_scaling');
    }
  }

  // Weekend
  if (ctx.isWeekend && ctx.weekendLeverageMultiplier) {
    maxLev = Math.floor(maxLev * ctx.weekendLeverageMultiplier);
    constraints.push('weekend_mode');
  }

  // Regime multiplier
  if (ctx.regimeLeverageMultiplier && ctx.regimeLeverageMultiplier < 1) {
    maxLev = Math.max(1, Math.round(maxLev * ctx.regimeLeverageMultiplier));
    constraints.push('regime_reduction');
  }

  // Recommended = 30%-60% of max
  const recLevMin = Math.max(1, Math.round(maxLev * 0.3));
  const recLevMax = Math.round(maxLev * 0.6);
  const recSizeMin = Math.max(5, Math.round(maxSize * 0.2));
  const recSizeMax = Math.round(maxSize * 0.5);

  // Blocked pairs
  const blockedPairs = portfolio.positions.map(p => `${p.pair}:${p.side}`);

  return {
    maxLeverage: maxLev,
    recommendedLeverage: [recLevMin, recLevMax],
    maxSizePct: maxSize,
    recommendedSizePct: [recSizeMin, recSizeMax],
    minConfidence: this.config.minConfidence ?? 55,
    blockedPairs,
    constraints,
  };
}
```

## Task 5: Attribution Aggregation Views (#33)

**Files:**
- Create: Supabase migration for views

**Step 1: Create SQL views**

```sql
-- Win rate and PnL by regime
CREATE OR REPLACE VIEW attribution_by_regime AS
SELECT
  e.regime_at_entry AS regime,
  COUNT(*) AS total_trades,
  COUNT(*) FILTER (WHERE c.pnl_usd > 0) AS wins,
  ROUND(COUNT(*) FILTER (WHERE c.pnl_usd > 0)::NUMERIC / NULLIF(COUNT(*), 0) * 100, 1) AS win_rate_pct,
  ROUND(SUM(c.pnl_usd)::NUMERIC, 2) AS total_pnl_usd,
  ROUND(AVG(c.pnl_pct)::NUMERIC, 2) AS avg_pnl_pct,
  ROUND(AVG(c.holding_time_minutes)::NUMERIC, 0) AS avg_hold_minutes,
  ROUND(SUM(c.pnl_usd) FILTER (WHERE c.pnl_usd > 0)::NUMERIC /
    NULLIF(ABS(SUM(c.pnl_usd) FILTER (WHERE c.pnl_usd < 0)), 0)::NUMERIC, 2) AS profit_factor
FROM trade_executions e
JOIN trade_closes c ON c.execution_id = e.id
WHERE e.regime_at_entry IS NOT NULL
GROUP BY e.regime_at_entry;

-- Swarm vs single-agent comparison
CREATE OR REPLACE VIEW attribution_swarm_vs_single AS
SELECT
  CASE WHEN e.was_swarm THEN 'swarm' ELSE 'single' END AS mode,
  COUNT(*) AS total_trades,
  COUNT(*) FILTER (WHERE c.pnl_usd > 0) AS wins,
  ROUND(COUNT(*) FILTER (WHERE c.pnl_usd > 0)::NUMERIC / NULLIF(COUNT(*), 0) * 100, 1) AS win_rate_pct,
  ROUND(SUM(c.pnl_usd)::NUMERIC, 2) AS total_pnl_usd,
  ROUND(AVG(c.pnl_pct)::NUMERIC, 2) AS avg_pnl_pct
FROM trade_executions e
JOIN trade_closes c ON c.execution_id = e.id
WHERE e.was_swarm IS NOT NULL
GROUP BY e.was_swarm;

-- Exit reason breakdown
CREATE OR REPLACE VIEW attribution_by_exit_reason AS
SELECT
  c.exit_reason,
  COUNT(*) AS total,
  ROUND(AVG(c.pnl_pct)::NUMERIC, 2) AS avg_pnl_pct,
  ROUND(AVG(c.holding_time_minutes)::NUMERIC, 0) AS avg_hold_minutes
FROM trade_closes c
WHERE c.exit_reason IS NOT NULL
GROUP BY c.exit_reason;
```
